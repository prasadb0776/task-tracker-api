import importlib

import pytest

from app import config


@pytest.fixture
def reload_config():
    original_path = config.DATABASE_PATH

    yield

    importlib.reload(config)
    config.DATABASE_PATH = original_path


def test_database_path_uses_environment_variable(
    monkeypatch,
    reload_config,
):
    monkeypatch.setenv("DATABASE_PATH", "/tmp/custom_tasks.db")

    importlib.reload(config)

    assert config.DATABASE_PATH == "/tmp/custom_tasks.db"


def test_database_path_defaults_when_environment_variable_is_missing(
    monkeypatch,
    reload_config,
):
    monkeypatch.delenv("DATABASE_PATH", raising=False)

    importlib.reload(config)

    assert config.DATABASE_PATH == "tasks.db"

import sqlite3

import pytest

from app.database import database_connection


def test_database_connection_closes_after_exception(tmp_path):
    db_path = tmp_path / "connection_test.db"
    connection = None

    with pytest.raises(RuntimeError, match="simulated failure"):
        with database_connection(str(db_path)) as connection:
            connection.execute("CREATE TABLE example (id INTEGER)")
            raise RuntimeError("simulated failure")

    assert connection is not None

    with pytest.raises(sqlite3.ProgrammingError, match="closed database"):
        connection.execute("SELECT 1")

def test_delete_task_success(client):

from datetime import date, timedelta


def test_stats_for_empty_database(client):
    response = client.get("/tasks/stats")

    assert response.status_code == 200
    assert response.json() == {
        "total": 0,
        "todo": 0,
        "in_progress": 0,
        "done": 0,
        "overdue": 0,
    }


def test_stats_count_tasks_by_status(client):
    client.post("/tasks", json={"title": "Todo task"})
    client.post(
        "/tasks",
        json={"title": "Active task", "status": "in_progress"},
    )
    client.post(
        "/tasks",
        json={"title": "Finished task", "status": "done"},
    )

    response = client.get("/tasks/stats")

    assert response.status_code == 200
    assert response.json() == {
        "total": 3,
        "todo": 1,
        "in_progress": 1,
        "done": 1,
        "overdue": 0,
    }


def test_stats_count_overdue_tasks(client):
    yesterday = (date.today() - timedelta(days=1)).isoformat()

    todo = client.post(
        "/tasks",
        json={"title": "Overdue todo"},
    ).json()

    active = client.post(
        "/tasks",
        json={"title": "Overdue active", "status": "in_progress"},
    ).json()

    completed = client.post(
        "/tasks",
        json={"title": "Completed task", "status": "done"},
    ).json()

    # Set past due dates after creation.
    for task in (todo, active, completed):
        response = client.patch(
            f"/tasks/{task['id']}",
            json={"due_date": yesterday},
        )
        assert response.status_code == 200

    response = client.get("/tasks/stats")

    assert response.status_code == 200
    data = response.json()
    assert data["total"] == 3
    assert data["todo"] == 1
    assert data["in_progress"] == 1
    assert data["done"] == 1
    assert data["overdue"] == 2

def test_completing_task_sets_completed_at(client):
    created = client.post(
        "/tasks",
        json={"title": "Finish testing"},
    ).json()

    response = client.patch(
        f"/tasks/{created['id']}",
        json={"status": "done"},
    )

    assert response.status_code == 200
    data = response.json()
    assert data["status"] == "done"
    assert data["completed_at"] is not None


def test_reopening_task_clears_completed_at(client):
    created = client.post(
        "/tasks",
        json={"title": "Complete and reopen"},
    ).json()

    completed = client.patch(
        f"/tasks/{created['id']}",
        json={"status": "done"},
    )
    assert completed.status_code == 200
    assert completed.json()["completed_at"] is not None

    reopened = client.patch(
        f"/tasks/{created['id']}",
        json={"status": "todo"},
    )

    assert reopened.status_code == 200
    assert reopened.json()["status"] == "todo"
    assert reopened.json()["completed_at"] is None

def test_put_replaces_task_fields(client):
    created = client.post(
        "/tasks",
        json={
            "title": "Old title",
            "description": "Old description",
            "priority": "low",
        },
    ).json()

    response = client.put(
        f"/tasks/{created['id']}",
        json={
            "title": "Replacement title",
            "description": "New description",
            "priority": "high",
            "status": "todo",
        },
    )

    assert response.status_code == 200
    data = response.json()
    assert data["title"] == "Replacement title"
    assert data["description"] == "New description"
    assert data["priority"] == "high"


def test_put_resets_omitted_fields_to_defaults(client):
    created = client.post(
        "/tasks",
        json={
            "title": "Original task",
            "description": "Original description",
            "status": "in_progress",
            "priority": "high",
        },
    )
    task_id = created.json()["id"]

    response = client.put(
        f"/tasks/{task_id}",

    assert data["title"] == "Replaced task"
    assert data["description"] is None
    assert data["status"] == "todo"
    assert data["priority"] == "medium"
